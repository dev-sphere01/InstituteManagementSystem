    if (paramName.includes('employee') && paramName.includes('staff development')) return review.EmployeeStaffDevelopment
    if (paramName.includes('decision making')) return review.DecisionMaking
    if (paramName.includes('productivity')) return review.ProductivityEfficiency
    if (paramName.includes('call for duty')) return review.CallForDuty
    if (paramName.includes('attendance')) return review.AttendancePunctuality
    if (paramName.includes('behaviour')) return review.BehaviourOverallDiscipline
    if (paramName.includes('integrity') && !paramName.includes('loyalty')) return review.Integrity
    if (paramName.includes('loyalty')) return review.LoyaltySubmissionToCompanyGoals

    return 0

    const cat2Score = calculateCategoryScore(empId, 2) // Discipline - 20%
    const cat3Score = calculateCategoryScore(empId, 3) // Integrity - 20%

    // Calculate weighted score: (K4*0.6+N4*0.2+Q4*0.2)/4*100
    const finalScore = ((cat1Score * 0.60) + (cat2Score * 0.20) + (cat3Score * 0.20)) / 4 * 100

    return Math.round(finalScore * 100) / 100

                {/* <div className="grid grid-cols-2">
                  <h2 className="text-xl font-semibold text-gray-800 mb-4">Review Details</h2>
                </div> */}
                <div className="space-y-4">
                  {/* Review Date */}
                  <div className="pb-2 border-b border-gray-200 grid grid-cols-2">
                    <div>
                      <div className="text-sm text-gray-500">Submitted On</div>
                      <div className="font-medium text-gray-800">
                        {new Date(selectedReviewDetail.SubmittedAt).toLocaleDateString('en-US', {
                          year: 'numeric',
                          month: 'long',
                          day: 'numeric'
                        })}
                      </div>
                      <div className="text-sm text-gray-500 mt-1">
                        {selectedReviewDetail.Month}/{selectedReviewDetail.Year}
                      </div>
                    </div>
                    {/* Final Score */}
                    <div className="bg-gradient-to-r from-green-50 to-blue-50 rounded-lg p-2 border border-green-200 grid grid-cols-2 items-center">
                      <div className="text-4xl font-bold text-green-600 text-center">
                        {selectedReviewDetail.FinalPercentage}%
                      </div>
                      <div className="text-sm text-gray-600 ms-2 leading-tight text-left">
                        Final <br /> Performance <br /> Score
                      </div>
                    </div>

                  </div>


                  {/* Performance Parameters */}
                  <div className="space-y-3">
                    <h3 className="font-semibold text-gray-700 text-sm uppercase">Performance (60%)</h3>
                    <div className="grid grid-cols-3 gap-3 text-sm">
                      <div className="bg-gray-50 p-2  border-gray-300 rounded border">
                        <div className="text-gray-600 text-xs">Self Development</div>
                        <div className="font-semibold text-gray-800">{selectedReviewDetail.SelfDevelopment}/4</div>
                      </div>
                      <div className="bg-gray-50 p-2  border-gray-300 rounded border">
                        <div className="text-gray-600 text-xs">Staff Development</div>
                        <div className="font-semibold text-gray-800">{selectedReviewDetail.EmployeeStaffDevelopment}/4</div>
                      </div>
                      <div className="bg-gray-50 p-2  border-gray-300 rounded border">
                        <div className="text-gray-600 text-xs">Decision Making</div>
                        <div className="font-semibold text-gray-800">{selectedReviewDetail.DecisionMaking}/4</div>
                      </div>
                      <div className="bg-gray-50 p-2  border-gray-300 rounded border">
                        <div className="text-gray-600 text-xs">Productivity</div>
                        <div className="font-semibold text-gray-800">{selectedReviewDetail.ProductivityEfficiency}/4</div>
                      </div>
                      <div className="bg-gray-50 p-2  border-gray-300 rounded border">
                        <div className="text-gray-600 text-xs">Call for Duty</div>
                        <div className="font-semibold text-gray-800">{selectedReviewDetail.CallForDuty}/4</div>
                      </div>
                      <div className="bg-blue-100 p-2  border-gray-300 rounded border">
                        <div className="text-blue-700 text-xs font-medium">Score</div>
                        <div className="font-bold text-blue-800">{selectedReviewDetail.PerformanceScore}</div>
                      </div>
                    </div>
                  </div>

                  {/* Discipline Parameters */}
                  <div className="space-y-3">
                    <h3 className="font-semibold text-gray-700 text-sm uppercase">Discipline (20%)</h3>
                    <div className="grid grid-cols-3 gap-3 text-sm">
                      <div className="bg-gray-50 p-2  border-gray-300 rounded border">
                        <div className="text-gray-600 text-xs">Attendance</div>
                        <div className="font-semibold text-gray-800">{selectedReviewDetail.AttendancePunctuality}/4</div>
                      </div>
                      <div className="bg-gray-50 p-2  border-gray-300 rounded border">
                        <div className="text-gray-600 text-xs">Behaviour</div>
                        <div className="font-semibold text-gray-800">{selectedReviewDetail.BehaviourOverallDiscipline}/4</div>
                      </div>
                      <div className="bg-orange-100 p-2  border-gray-300 rounded border">
                        <div className="text-orange-700 text-xs font-medium">Score</div>
                        <div className="font-bold text-orange-800">{selectedReviewDetail.DisciplineScore}</div>
                      </div>
                    </div>
                  </div>

                  {/* Integrity Parameters */}
                  <div className="space-y-3">
                    <h3 className="font-semibold text-gray-700 text-sm uppercase">Integrity (20%)</h3>
                    <div className="grid grid-cols-3 gap-3 text-sm">
                      <div className="bg-gray-50 p-2  border-gray-300 rounded border">
                        <div className="text-gray-600 text-xs">Integrity</div>
                        <div className="font-semibold text-gray-800">{selectedReviewDetail.Integrity}/4</div>
                      </div>
                      <div className="bg-gray-50 p-2  border-gray-300 rounded border">
                        <div className="text-gray-600 text-xs">Loyalty</div>
                        <div className="font-semibold text-gray-800">{selectedReviewDetail.LoyaltySubmissionToCompanyGoals}/4</div>
                      </div>
                      <div className="bg-purple-100 p-2  border-gray-300 rounded border">
                        <div className="text-purple-700 text-xs font-medium">Score</div>
                        <div className="font-bold text-purple-800">{selectedReviewDetail.IntegrityScore}</div>
                      </div>
                    </div>
                  </div>
                </div>
              </motion.div>
            </div>
          )}
        </div>
      )}

      {/* Empty States when no session is selected */}
      {!selectedSessionId && (
        <motion.div
          initial={{ opacity: 0, y: 20 }}
          animate={{ opacity: 1, y: 0 }}
          className="bg-white rounded-xl shadow-sm border border-gray-200 p-12"
        >
          <div className="text-center">
            <FolderOpen className="h-16 w-16 mx-auto text-gray-300 mb-4" />
            <h3 className="text-lg font-semibold text-gray-800 mb-2">No Review Session Selected</h3>
            <p className="text-gray-500 max-w-md mx-auto">
              Please select a review session from the dropdown above to begin performance assessments and access all features.
            </p>
          </div>
        </motion.div>
      )}
    </div>
  )
}
